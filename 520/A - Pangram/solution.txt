package main
 
import (
	"fmt"
	"strings"
)
 
func main() {
	var a int
	var s string
	fmt.Scan(&a)
	fmt.Scan(&s)
	if a < 26 {
		fmt.Println("NO")
	} else {
		s = strings.ToLower(s)
		m := make(map[rune]bool)
		for _, c := range s {
			m[c] = true
		}
		if len(m) == 26 {
			fmt.Println("YES")
		} else {
			fmt.Println("NO")
		}
	}
}
