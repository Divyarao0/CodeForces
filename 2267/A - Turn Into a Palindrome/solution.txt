package main
 
import "fmt"
 
func main() {
	var t int
	fmt.Scan(&t)
 
	answers := make([]int, t)
	for k := 0; k < t; k++ {
		var n int
		var c, s string
		fmt.Scan(&n, &c, &s)
 
		coins := 0
		for i := 0; i < n/2; i++ {
			a, b := s[i], s[n-1-i]
			if a == b {
				continue
			}
			if a == c[0] || b == c[0] {
				coins += 1 // change the other one to c
			} else {
				coins += 2 // change both to c
			}
		}
		answers[k] = coins
	}
 
	// print all answers together after reading all input
	for _, ans := range answers {
		fmt.Println(ans)
	}
}
