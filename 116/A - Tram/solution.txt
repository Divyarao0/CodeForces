package main
 
import "fmt"
 
func main() {
	var n int
	fmt.Scan(&n)
 
	a := make([]int, n)
	b := make([]int, n)
	x := make([]int, n)
 
	for i := 0; i < n; i++ {
		fmt.Scan(&a[i], &b[i])
	}
 
	x[0] = b[0]
 
	for i := 1; i < n; i++ {
		x[i] = x[i-1] - a[i] + b[i]
	}
 
	max := x[0]
 
	for i := 1; i < n; i++ {
		if x[i] > max {
			max = x[i]
		}
	}
 
	fmt.Println(max)
}
