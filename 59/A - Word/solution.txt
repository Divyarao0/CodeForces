package main
 
import (
	"fmt"
	"strings"
)
 
func main() {
	var s string
	fmt.Scan(&s)
	uppercase := 0
	lowercase := 0
	for i := 0; i < len(s); i++ {
		ch := s[i]
		if ch >= 'a' && ch <= 'z' {
			lowercase++
		} else if ch >= 'A' && ch <= 'Z' {
			uppercase++
		}
	}
	if uppercase > lowercase {
		s := strings.ToUpper(s)
		fmt.Println(s)
	} else {
		s := strings.ToLower(s)
		fmt.Println(s)
	}
 
}
